// utilizada a mais. Nenhum veículo utiliza o estacionamento por um período igual ou superior a 24 horas.

função calculaDiferenca(hE, mE, hS, mS: inteiro) : inteiro
var
     minutosEnt, minutosSai: inteiro
início
     minutosEnt ← hE * 60 + mE
     se hS < hE então
          hS ← hS + 24
     fim se
     minutosSai ← hS * 60 + mS
     retorne minutosSai - minutosEnt
fim

função calculaValor(tempoMinutos: inteiro) : real
var
          horasExtras: inteiro

          se tempoMinutos <= 120 então
                retorne 10.00;
          senão
                horasExtras ← Math.ceil((tempoMinutos-120) div 60);
                retorne 10.00 + horasExtras * 2.50;

----------   Essas funções em JAVA:

static int calculaDiferenca(int hE, int mE, int hS, int mS) {

       int minutosEnt = hE * 60 + mE;

       if(hS < hE)

             hS = hS + 24;

       int minutosSai = hS * 60 + mS;

return minutosSai - minutosEnt;

}

static double calculaValor(int tempoMinutos) {

       if(tempoMinutos <= 120)

             return 10.00;
else {
             int horasExtras = (int) Math.ceil((tempoMinutos-120) / 60.0);

             return 10.00 + horasExtras * 2.50;
       }
}
